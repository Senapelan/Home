// Fungsi utama untuk menyusun data keluarga
function sortDataKeluarga() {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    const lastRow = sheet.getLastRow();
    const lastColumn = sheet.getLastColumn();
    
    if (lastRow <= 1) {
      SpreadsheetApp.getUi().alert('Tidak ada data untuk disusun');
      return;
    }
    
    // Deteksi jumlah kolom yang ada
    const actualHeaders = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
    
    // Ambil semua data dengan format sebagai string
    const dataRange = sheet.getRange(1, 1, lastRow, lastColumn);
    const dataValues = dataRange.getValues();
    
    // Konversi semua nilai ke string untuk mencegah perubahan format
    for (let i = 0; i < dataValues.length; i++) {
      for (let j = 0; j < dataValues[i].length; j++) {
        // Jika nilainya adalah number, konversi ke string dengan format aslinya
        if (typeof dataValues[i][j] === 'number') {
          // Cek apakah ini kolom RT/RW atau NO TELP
          const header = actualHeaders[j];
          if (header === 'RT' || header === 'RW') {
            // Format 3 digit dengan leading zero
            dataValues[i][j] = dataValues[i][j].toString().padStart(3, '0');
          } else if (header === 'NOMOR TELEPON') {
            // Format nomor telepon tanpa leading zero diubah
            dataValues[i][j] = dataValues[i][j].toString();
          } else {
            // Untuk kolom lain, konversi biasa
            dataValues[i][j] = dataValues[i][j].toString();
          }
        } else if (dataValues[i][j] && typeof dataValues[i][j] === 'string') {
          // Jika sudah string, pastikan format RT/RW tetap 3 digit
          const header = actualHeaders[j];
          if (header === 'RT' || header === 'RW') {
            // Pastikan selalu 3 digit dengan leading zero
            if (dataValues[i][j].match(/^\d+$/)) {
              dataValues[i][j] = dataValues[i][j].toString().padStart(3, '0');
            }
          }
        }
      }
    }
    
    // Pisahkan header dan data
    const headers = dataValues[0];
    let dataRows = [];
    for (let i = 1; i < dataValues.length; i++) {
      dataRows.push({
        rowIndex: i + 1,
        values: dataValues[i]
      });
    }
    
    // Update KODE KELUARGA jika kolomnya ada
    const kodeKeluargaIndex = headers.indexOf('KODE KELUARGA');
    const nomorKKIndex = headers.indexOf('NOMOR KK');
    const statusHubunganIndex = headers.indexOf('STATUS HUBUNGAN DALAM KELUARGA');
    const tanggalLahirIndex = headers.indexOf('TANGGAL LAHIR');
    const kodeAnggotaIndex = headers.indexOf('KODE ANGGOTA KELUARGA');
    
    if (kodeKeluargaIndex === -1 || nomorKKIndex === -1) {
      SpreadsheetApp.getUi().alert('Struktur kolom tidak sesuai!');
      return;
    }
    
    // Generate ulang KODE KELUARGA yang benar
    for (let i = 0; i < dataRows.length; i++) {
      const row = dataRows[i].values;
      const nomorKK = row[nomorKKIndex] ? row[nomorKKIndex].toString() : '';
      if (nomorKK && nomorKK.length >= 16) {
        const uniquePart = nomorKK.substring(6, 12);
        row[kodeKeluargaIndex] = 'KK_' + uniquePart;
      }
    }
    
    // Sorting data
    dataRows.sort((a, b) => {
      let kodeA = a.values[kodeKeluargaIndex] || '';
      let kodeB = b.values[kodeKeluargaIndex] || '';
      
      if (kodeA < kodeB) return -1;
      if (kodeA > kodeB) return 1;
      
      if (statusHubunganIndex !== -1) {
        let statusA = a.values[statusHubunganIndex] || '';
        let statusB = b.values[statusHubunganIndex] || '';
        let priorityA = HUBUNGAN_PRIORITAS[statusA] || 99;
        let priorityB = HUBUNGAN_PRIORITAS[statusB] || 99;
        
        if (priorityA !== priorityB) {
          return priorityA - priorityB;
        }
        
        if (statusA === 'ANAK' && statusB === 'ANAK' && tanggalLahirIndex !== -1) {
          let tglA = parseTanggal(a.values[tanggalLahirIndex]);
          let tglB = parseTanggal(b.values[tanggalLahirIndex]);
          return tglA - tglB;
        }
      }
      
      return 0;
    });
    
    // Update NO URUT OTOMATIS jika kolomnya ada
    const nomorUrutIndex = headers.indexOf('NO URUT OTOMATIS');
    if (nomorUrutIndex !== -1) {
      let currentKodeKeluarga = '';
      let anggotaCounter = 1;
      
      for (let i = 0; i < dataRows.length; i++) {
        const row = dataRows[i].values;
        const kodeKeluarga = row[kodeKeluargaIndex];
        
        if (kodeKeluarga !== currentKodeKeluarga) {
          currentKodeKeluarga = kodeKeluarga;
          anggotaCounter = 1;
        }
        
        let nomorUrut = '';
        if (kodeKeluarga) {
          if (kodeAnggotaIndex !== -1 && row[kodeAnggotaIndex]) {
            const parts = row[kodeAnggotaIndex].split('_');
            if (parts.length >= 4) {
              nomorUrut = kodeKeluarga + '-' + parts[3];
            } else {
              nomorUrut = kodeKeluarga + '-' + anggotaCounter.toString().padStart(2, '0');
            }
          } else {
            nomorUrut = kodeKeluarga + '-' + anggotaCounter.toString().padStart(2, '0');
          }
        }
        
        row[nomorUrutIndex] = nomorUrut;
        anggotaCounter++;
      }
    }
    
    // Tulis ulang semua data dengan format yang benar
    const newData = [headers];
    for (let i = 0; i < dataRows.length; i++) {
      const row = [...dataRows[i].values]; // Salin array
      
      // Pastikan format kolom tertentu tetap terjaga
      for (let j = 0; j < row.length; j++) {
        const header = headers[j];
        
        // Format RT dan RW selalu 3 digit dengan leading zero
        if (header === 'RT' || header === 'RW') {
          if (row[j] && row[j].toString().match(/^\d+$/)) {
            row[j] = row[j].toString().padStart(3, '0');
          }
        }
        
        // Format NOMOR TELEPON sebagai string
        if (header === 'NOMOR TELEPON') {
          if (row[j] && typeof row[j] === 'number') {
            row[j] = row[j].toString();
          }
        }
      }
      
      newData.push(row);
    }
    
    // Hapus semua data lama dan tulis ulang
    sheet.clearContents();
    
    // Set nilai dengan format yang benar
    const outputRange = sheet.getRange(1, 1, newData.length, lastColumn);
    outputRange.setValues(newData);
    
    // Set format sel untuk kolom tertentu
    const rtIndex = headers.indexOf('RT');
    const rwIndex = headers.indexOf('RW');
    const telpIndex = headers.indexOf('NOMOR TELEPON');
    
    if (rtIndex !== -1) {
      const rtRange = sheet.getRange(2, rtIndex + 1, newData.length - 1, 1);
      rtRange.setNumberFormat('000'); // Format 3 digit
    }
    
    if (rwIndex !== -1) {
      const rwRange = sheet.getRange(2, rwIndex + 1, newData.length - 1, 1);
      rwRange.setNumberFormat('000'); // Format 3 digit
    }
    
    if (telpIndex !== -1) {
      const telpRange = sheet.getRange(2, telpIndex + 1, newData.length - 1, 1);
      telpRange.setNumberFormat('@'); // Format sebagai teks
    }
    
    // Format header
    sheet.getRange(1, 1, 1, lastColumn).setFontWeight('bold');
    sheet.getRange(1, 1, 1, lastColumn).setBackground('#667eea');
    sheet.getRange(1, 1, 1, lastColumn).setFontColor('#ffffff');
    
    SpreadsheetApp.getUi().alert('✅ Data berhasil disusun!\nTotal ' + dataRows.length + ' baris data.\nFormat RT/RW (3 digit) dan No. Telepon telah dipertahankan.');
    
  } catch (error) {
    SpreadsheetApp.getUi().alert('❌ Error: ' + error.toString());
  }
}

// Fungsi untuk update NO URUT OTOMATIS
function updateNomorUrut() {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    const lastRow = sheet.getLastRow();
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    
    const nomorUrutIndex = headers.indexOf('NO URUT OTOMATIS');
    const kodeKeluargaIndex = headers.indexOf('KODE KELUARGA');
    const kodeAnggotaIndex = headers.indexOf('KODE ANGGOTA KELUARGA');
    
    if (nomorUrutIndex === -1) {
      SpreadsheetApp.getUi().alert('Kolom NO URUT OTOMATIS belum ada. Tambah dulu dari menu.');
      return;
    }
    
    if (lastRow <= 1) return;
    
    // Ambil data
    const dataRange = sheet.getRange(2, 1, lastRow - 1, headers.length);
    const dataValues = dataRange.getValues();
    
    let currentKodeKeluarga = '';
    let anggotaCounter = 1;
    let nomorUrutValues = [];
    
    for (let i = 0; i < dataValues.length; i++) {
      const row = dataValues[i];
      const kodeKeluarga = kodeKeluargaIndex !== -1 ? row[kodeKeluargaIndex] : '';
      const kodeAnggota = kodeAnggotaIndex !== -1 ? row[kodeAnggotaIndex] : '';
      
      if (kodeKeluarga !== currentKodeKeluarga) {
        currentKodeKeluarga = kodeKeluarga;
        anggotaCounter = 1;
      }
      
      let nomorUrut = '';
      if (kodeKeluarga) {
        if (kodeAnggota) {
          const parts = kodeAnggota.split('_');
          if (parts.length >= 4) {
            nomorUrut = kodeKeluarga + '-' + parts[3];
          } else {
            nomorUrut = kodeKeluarga + '-' + anggotaCounter.toString().padStart(2, '0');
          }
        } else {
          nomorUrut = kodeKeluarga + '-' + anggotaCounter.toString().padStart(2, '0');
        }
      }
      
      nomorUrutValues.push([nomorUrut]);
      anggotaCounter++;
    }
    
    // Tulis NO URUT OTOMATIS
    sheet.getRange(2, nomorUrutIndex + 1, nomorUrutValues.length, 1).setValues(nomorUrutValues);
    
    SpreadsheetApp.getUi().alert('✅ NO URUT OTOMATIS berhasil diupdate!');
    
  } catch (error) {
    SpreadsheetApp.getUi().alert('❌ Error: ' + error.toString());
  }
}

// Fungsi hapus duplikat
function hapusDuplikat() {
  try {
    const ui = SpreadsheetApp.getUi();
    const response = ui.alert(
      '⚠️ Hapus Duplikat',
      'Yakin ingin menghapus data duplikat berdasarkan NIK?',
      ui.ButtonSet.YES_NO
    );
    
    if (response !== ui.Button.YES) return;
    
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    const lastRow = sheet.getLastRow();
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const nikIndex = headers.indexOf('NIK');
    
    if (nikIndex === -1) {
      ui.alert('Kolom NIK tidak ditemukan!');
      return;
    }
    
    if (lastRow <= 1) return;
    
    // Ambil data
    const dataRange = sheet.getRange(2, 1, lastRow - 1, headers.length);
    const dataValues = dataRange.getValues();
    
    // Cari duplikat berdasarkan NIK
    let seenNIK = new Set();
    let rowsToDelete = [];
    
    for (let i = dataValues.length - 1; i >= 0; i--) {
      const nik = dataValues[i][nikIndex];
      if (nik && seenNIK.has(nik)) {
        rowsToDelete.push(i + 2);
      } else {
        seenNIK.add(nik);
      }
    }
    
    // Hapus baris duplikat
    for (let i = rowsToDelete.length - 1; i >= 0; i--) {
      sheet.deleteRow(rowsToDelete[i]);
    }
    
    ui.alert('✅ ' + rowsToDelete.length + ' data duplikat berhasil dihapus!');
    
  } catch (error) {
    SpreadsheetApp.getUi().alert('❌ Error: ' + error.toString());
  }
}

// Fungsi untuk parse tanggal
function parseTanggal(tanggalStr) {
  if (!tanggalStr) return new Date(2100, 0, 1);
  
  const parts = tanggalStr.split('/');
  if (parts.length === 3) {
    const day = parseInt(parts[0], 10);
    const month = parseInt(parts[1], 10) - 1;
    const year = parseInt(parts[2], 10);
    return new Date(year, month, day);
  }
  
  return new Date(2100, 0, 1);
}

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  try {
    const sheet = getOrCreateSheet();
    
    let data = null;
    if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    }
    
    if (e.parameter.action === 'get') {
      return getData(sheet, e.parameter.entry);
    } else if (data && data.action === 'update') {
      return updateData(sheet, data);
    } else if (data && data.action === 'delete') {
      return deleteData(sheet, data);
    } else if (data) {
      return saveData(sheet, data);
    }
    
    return sendResponse('success', 'Sheet siap digunakan');
    
  } catch (error) {
    return sendResponse('error', error.toString());
  }
}

function getOrCreateSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);
  
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.getRange(1, 1, 1, BASE_HEADERS.length).setValues([BASE_HEADERS]);
    sheet.getRange(1, 1, 1, BASE_HEADERS.length).setFontWeight('bold');
    sheet.getRange(1, 1, 1, BASE_HEADERS.length).setBackground('#667eea');
    sheet.getRange(1, 1, 1, BASE_HEADERS.length).setFontColor('#ffffff');
  }
  
  return sheet;
}

function saveData(sheet, data) {
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  const headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
  
  // Buat array data sesuai dengan kolom yang ada
  let newRowData = [];
  for (let i = 0; i < headers.length; i++) {
    const header = headers[i];
    switch(header) {
      case 'TIMESTAMP': newRowData.push(data.timestamp || ''); break;
      case 'KODE KELUARGA': newRowData.push(data.kodeKeluarga || ''); break;
      case 'NOMOR KK': newRowData.push(data.nomorKK || ''); break;
      case 'NAMA KEPALA KELUARGA': newRowData.push(data.namaKepalaKeluarga || ''); break;
      case 'ALAMAT': newRowData.push(data.alamat || ''); break;
      case 'RT': newRowData.push(data.rt || ''); break;
      case 'RW': newRowData.push(data.rw || ''); break;
      case 'KELURAHAN/DESA': newRowData.push(data.kelurahan || ''); break;
      case 'KECAMATAN': newRowData.push(data.kecamatan || ''); break;
      case 'KABUPATEN/KOTA': newRowData.push(data.kabupaten || ''); break;
      case 'KODE POS': newRowData.push(data.kodePos || ''); break;
      case 'PROVINSI': newRowData.push(data.provinsi || ''); break;
      case 'KODE ANGGOTA KELUARGA': newRowData.push(data.kodeAnggota || ''); break;
      case 'NAMA LENGKAP': newRowData.push(data.namaLengkap || ''); break;
      case 'NIK': newRowData.push(data.nik || ''); break;
      case 'JENIS KELAMIN': newRowData.push(data.jenisKelamin || ''); break;
      case 'TEMPAT LAHIR': newRowData.push(data.tempatLahir || ''); break;
      case 'TANGGAL LAHIR': newRowData.push(data.tanggalLahir || ''); break;
      case 'AGAMA': newRowData.push(data.agama || ''); break;
      case 'PENDIDIKAN': newRowData.push(data.pendidikan || ''); break;
      case 'PEKERJAAN': newRowData.push(data.pekerjaan || ''); break;
      case 'STATUS PERKAWINAN': newRowData.push(data.statusPerkawinan || ''); break;
      case 'STATUS HUBUNGAN DALAM KELUARGA': newRowData.push(data.statusHubungan || ''); break;
      case 'KEWARGANEGARAAN': newRowData.push(data.kewarganegaraan || ''); break;
      case 'NO PASPOR': newRowData.push(data.noPaspor || ''); break;
      case 'NO KITAS/KITAP': newRowData.push(data.noKitas || ''); break;
      case 'NAMA AYAH': newRowData.push(data.namaAyah || ''); break;
      case 'NAMA IBU': newRowData.push(data.namaIbu || ''); break;
      case 'SUKU': newRowData.push(data.suku || 'TIDAK DIKETAHUI'); break;
      case 'STATUS KEKAYAAN': newRowData.push(data.statusKekayaan || 'TIDAK DIKETAHUI'); break;
      case 'STATUS KESEHATAN': newRowData.push(data.statusKesehatan || 'TIDAK DIKETAHUI'); break;
      case 'RIWAYAT PENYAKIT': newRowData.push(data.riwayatPenyakit || 'TIDAK DIKETAHUI'); break;
      case 'JENIS PENYAKIT': newRowData.push(data.jenisPenyakit || ''); break;
      case 'STATUS KEBERSIHAN': newRowData.push(data.statusKebersihan || 'TIDAK DIKETAHUI'); break;
      case 'NOMOR TELEPON': newRowData.push(data.nomorTelepon || ''); break;
      case 'ALAMAT EMAIL': newRowData.push(data.email || ''); break;
      case 'NO URUT OTOMATIS':
        // Generate nomor urut
        let nomorUrut = '';
        if (data.kodeKeluarga) {
          let urutan = '99';
          if (data.kodeAnggota) {
            const parts = data.kodeAnggota.split('_');
            if (parts.length >= 4) urutan = parts[3];
          } else {
            const order = HUBUNGAN_ORDER[data.statusHubungan] || '99';
            urutan = order;
          }
          nomorUrut = data.kodeKeluarga + '-' + urutan;
        }
        newRowData.push(nomorUrut);
        break;
      default:
        newRowData.push('');
    }
  }
  
  sheet.appendRow(newRowData);
  return sendResponse('success', 'Data berhasil disimpan');
}

function updateData(sheet, data) {
  const rowIndex = data.rowIndex;
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  
  if (rowIndex > lastRow) {
    return sendResponse('error', 'Data tidak ditemukan');
  }
  
  // Hapus baris lama
  sheet.deleteRow(rowIndex);
  
  // Insert ulang
  return saveData(sheet, data);
}

function deleteData(sheet, data) {
  const rowIndex = data.rowIndex;
  sheet.deleteRow(rowIndex);
  return sendResponse('success', 'Data berhasil dihapus');
}

function getData(sheet, entry) {
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  
  if (lastRow <= 1) {
    return sendResponse('success', []);
  }
  
  let limit = entry === 'all' ? lastRow - 1 : parseInt(entry);
  limit = Math.min(limit, lastRow - 1);
  
  const headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
  const dataRange = sheet.getRange(2, 1, limit, lastColumn);
  const dataValues = dataRange.getValues();
  
  const result = [];
  for (let i = 0; i < dataValues.length; i++) {
    const row = {};
    for (let j = 0; j < headers.length; j++) {
      row[headers[j]] = dataValues[i][j];
    }
    result.push(row);
  }
  
  return sendResponse('success', result);
}

function sendResponse(status, data) {
  return ContentService
    .createTextOutput(JSON.stringify({
      status: status,
      data: data
    }))
    .setMimeType(ContentService.MimeType.JSON);
}
